45CBackend::OnInitReturnValue CBackendFreeRDPServer::OnInit()
46{
47 shadow_subsystem_set_entry_builtin(NULL);
48
49 m_pServer = shadow_server_new();
50 if (!m_pServer)
51 {
52 qCritical(log) << "Server new failed";
53 return OnInitReturnValue::Fail;
54 }
55
56 SetParameters();
57
58 int nRet = shadow_server_init(m_pServer);
59 if(nRet < 0)
60 {
61 qCritical(log) << "Server initialization failed:" << nRet;
62 return OnInitReturnValue::Fail;
63 }
64 m_bServerInit = true;
65
66 if ((nRet = shadow_server_start(m_pServer)) < 0)
67 {
68 qCritical(log) << "Failed to start server:" << nRet;
69 return OnInitReturnValue::Fail;
70 }
71
72 // Don't use OnProcess (qt event loop)
73 // Because freerdp has new thread process loop
74 return OnInitReturnValue::NotUseOnProcess;
75}
