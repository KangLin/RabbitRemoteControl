135bool CBackendFtpServer::onFilter(QSslSocket *socket)
136{
137 bool bFilte = false;
138 if(!socket) return true;
139 QString szIP = socket->peerAddress().toString();
140
141 auto& white = m_pPara->m_WhiteFilter;
142 auto& black = m_pPara->m_BlackFilter;
143
144 bool bInWhite = false;
145 if(!white.isEmpty()) {
146 white.OnProcess([&](const QString& szKey)->int {
147 QHostAddress addr(szIP);
148 auto sub = QHostAddress::parseSubnet(szKey);
149 if(addr.isInSubnet(sub.first, sub.second)) {
150 bInWhite = true;
151 return -1;
152 }
153 return 0;
154 }, true);
155 }
156
157 if(!bInWhite && !black.isEmpty()) {
158 black.OnProcess([&](const QString& szKey)->int {
159 QHostAddress addr(szIP);
160 auto sub = QHostAddress::parseSubnet(szKey);
161 if(addr.isInSubnet(sub.first, sub.second)) {
162 qInfo(log) << "Filtered" << szIP << "in blacklist";
163 bFilte = true;
164 return -1;
165 }
166 return 0;
167 }, true);
168 }
169
170 if(bFilte) return true;
171
172 if(m_pPara->GetConnectCount() >= 0 && m_pPara->GetConnectCount() <= m_Sockets.size()) {
173 qDebug(log) << "Exceeded the allowed number of connections:" << m_pPara->GetConnectCount();
174 return true;
175 }
176
177 bool check = false;
178 check = connect(socket, SIGNAL(disconnected()),
179 this, SLOT(slotDisconnected()));
180 Q_ASSERT(check);
181
182 quint16 port = socket->peerPort();
183 m_Sockets.append(socket);
184 emit sigConnected(szIP, port);
185 qDebug(log) << "Current connect count:" << m_Sockets.size()
186 << "; new connect from:" << szIP + ":" + QString::number(port);
187 return false;
188}
