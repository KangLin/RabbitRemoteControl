40CBackend::OnInitReturnValue CBackendFreeRDPServer::OnInit()
41{
42 shadow_subsystem_set_entry_builtin(NULL);
43
44 m_pServer = shadow_server_new();
45 if (!m_pServer)
46 {
47 qCritical(log) << "Server new failed";
48 return OnInitReturnValue::Fail;
49 }
50
51 SetParameters();
52
53 int nRet = shadow_server_init(m_pServer);
54 if(nRet < 0)
55 {
56 qCritical(log) << "Server initialization failed:" << nRet;
57 return OnInitReturnValue::Fail;
58 }
59 m_bServerInit = true;
60
61 if ((nRet = shadow_server_start(m_pServer)) < 0)
62 {
63 qCritical(log) << "Failed to start server:" << nRet;
64 return OnInitReturnValue::Fail;
65 }
66
67 // Don't use OnProcess (qt event loop)
68 // Because freerdp has new thread process loop
69 return OnInitReturnValue::NotUseOnProcess;
70}
