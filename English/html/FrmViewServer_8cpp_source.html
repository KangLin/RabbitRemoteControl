91void CFrmViewServer::slotDisconnected(const QString& szIp, const quint16 port)
92{
93 for (int row = 0; row < m_ModelConnect.rowCount(); row++) {
94 QModelIndex index = m_ModelConnect.index(row, 0);
95 QString ip = m_ModelConnect.data(index).toString();
96 QModelIndex indexPort = m_ModelConnect.index(row, 1);
97 QString szPort = m_ModelConnect.data(indexPort).toString();
98 if (szIp == ip && szPort == QString::number(port)) {
99 m_ModelConnect.removeRows(row, 1);
100 m_Stats.Disconnects();
101 m_plbConnectCount->setText(m_Stats.ToString());
102 break;
103 }
104 }
105}

107void CFrmViewServer::slotContextMenuRequested(const QPoint& pos)
108{
109 if(m_ModelConnect.rowCount() == 0)
110 return;
111 QMenu m;
112 m.addAction(tr("Disconnect"), this, [this](){
113 QItemSelectionModel* pSelect = m_ptvClients->selectionModel();
114 QModelIndexList lstIndex = pSelect->selectedRows();
115 foreach(auto index, lstIndex) {
116 QString szIp = m_ModelConnect.item(index.row(), 0)->text();
117 quint16 port = m_ModelConnect.item(index.row(), 1)->text().toUInt();
118 emit sigDisconnect(szIp, port);
119 }
120 });
121 QPoint p = m_ptvClients->mapToGlobal(pos);
122 m.exec(p);
